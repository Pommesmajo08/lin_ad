cmake_minimum_required(VERSION 3.20)
project(LinadLogon LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(linad-credential-provider SHARED provider.cpp exports.def)
target_compile_definitions(linad-credential-provider PRIVATE UNICODE _UNICODE)
target_link_libraries(linad-credential-provider PRIVATE ole32 uuid secur32 advapi32)
add_executable(linad-logon-probe probe.cpp)
target_compile_definitions(linad-logon-probe PRIVATE UNICODE _UNICODE)
target_link_libraries(linad-logon-probe PRIVATE ole32 uuid secur32 advapi32 userenv)
if(MSVC)
  target_compile_options(linad-credential-provider PRIVATE /W4 /WX /utf-8)
  target_compile_options(linad-logon-probe PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(linad-credential-provider PRIVATE -Wall -Wextra -Werror)
  target_compile_options(linad-logon-probe PRIVATE -Wall -Wextra -Werror)
  target_link_options(linad-credential-provider PRIVATE -static)
  target_link_options(linad-logon-probe PRIVATE -static -municode)
endif()
